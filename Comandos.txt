-- Estrutura da tabela
DESCRIBE items;

-- Total de registros
SELECT COUNT(*) as total_items FROM items;

-- Últimos registros
SELECT codigo, nome_do_produto, preco, created_at 
FROM items 
ORDER BY created_at DESC 
LIMIT 5;

-- Por categoria
SELECT codigo_da_categoria, COUNT(*) as quantidade 
FROM items 
GROUP BY codigo_da_categoria 
ORDER BY quantidade DESC;

-- Range de preços
SELECT 
    MIN(preco) as preco_minimo,
    MAX(preco) as preco_maximo,
    AVG(preco) as preco_medio,
    COUNT(*) as total_com_preco
FROM items 
WHERE preco > 0;
